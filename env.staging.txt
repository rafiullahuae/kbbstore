# ---------------------------------------------------------------------------
# KBB staging — easywebsol.com/kbb-upgrade/
#
# Three apps now share this domain: the Console, the WordPress test site at
# /kb/, and this. The settings below keep them from colliding — in particular
# the session cookie, which is the one that causes confusing "randomly logged
# out" bugs when two Laravel apps share a hostname.
# ---------------------------------------------------------------------------

APP_NAME="K-Beauty Bliss"
APP_ENV=staging
APP_KEY=
APP_DEBUG=false
APP_URL=https://easywebsol.com/kbb-upgrade

# Prefixes every internal link. Leave EMPTY in production so the URL Contract
# resolves at the domain root exactly as it does today.
KBB_BASE_PATH=/kbb-upgrade

# Keep staging out of the index. Production must not carry this.
KBB_NOINDEX=true

# --- Database: its own, never shared with the Console -----------------------
DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=
DB_USERNAME=
DB_PASSWORD=

# --- Sessions ---------------------------------------------------------------
# A distinct cookie name AND path. Without both, this app and the Console
# overwrite each other's sessions on the shared hostname.
SESSION_DRIVER=database
SESSION_COOKIE=kbb_upgrade_session
SESSION_PATH=/kbb-upgrade
SESSION_SAME_SITE=lax
SESSION_SECURE_COOKIE=true

# --- Shared hosting: no Redis, no daemons -----------------------------------
CACHE_STORE=file
QUEUE_CONNECTION=database

# --- Mail -------------------------------------------------------------------
MAIL_MAILER=smtp
MAIL_HOST=
MAIL_PORT=587
MAIL_USERNAME=
MAIL_PASSWORD=
MAIL_ENCRYPTION=tls
MAIL_FROM_ADDRESS=info@kbeautybliss.com
MAIL_FROM_NAME="K-Beauty Bliss"

# --- Payments: staging keys only. Never paste these into chat (D-41). -------
STRIPE_KEY=
STRIPE_SECRET=
TABBY_PUBLIC_KEY=
TABBY_SECRET_KEY=
TAMARA_TOKEN=
